#+TITLE: Emacs Configuration
#+PROPERTY: header-args:emacs-lisp :tangle init.el :results silent :comments no

A literate configuration for Emacs 31, run as a daemon and used through
=emacsclient=. Nix provides every package, so this file only declares and
configures them. Blocks under =Early init= tangle to =early-init.el=; all
others tangle to =init.el=. Blocks are kept small so any one of them can be
evaluated alone to preview a change; to keep a change, edit this file and
rebuild.

Layout:

- =Early init=: settings applied before the first frame exists.
- =Core=: paths, defaults, quiet messages, session state and performance.
- =Appearance=: interface, theme and mode line.
- =Functions=: shared helper commands.
- =Editing=: modal editing, completion and general editing tools.
- =Programming=: language servers, formatting, tree-sitter and LaTeX.
- =Org=: helpers, SVG tags, core setup, agenda, capture and Roam.
- =Keybindings=: which-key, leader key and per-mode bindings.

* Early init
:PROPERTIES:
:header-args:emacs-lisp: :tangle early-init.el
:END:

Runs before the first frame exists, so it only holds settings that must be in
place before startup: package handling, the garbage collector, native
compilation and the look of the first frame.

** Startup

Turns off =package.el= (Nix owns the packages) and lifts the garbage collector
limit while loading. =gcmh= takes over afterwards.

#+begin_src emacs-lisp
  ;;; early-init.el --- Early startup -*- lexical-binding: t; -*-

  (require 'xdg)

  (setq package-enable-at-startup nil
        gc-cons-threshold most-positive-fixnum
        inhibit-compacting-font-caches t)
#+end_src

** Performance

** Theme

The theme follows the system. Colors are read from a JSON file written by an
external tool, and every face is derived from that palette. Missing or invalid
colors fall back to a built-in dark palette, so a broken file never breaks
startup.

*** Palette

Turns the parsed JSON file into a plist of semantic colors. =my/with-palette=
binds those names as variables so face lists can refer to them directly.

#+begin_src emacs-lisp
  (deftheme my-system)

  (defconst my/theme-fallback
    '(("BASE00" . "#1E1E2E") ("BASE01" . "#181825") ("BASE02" . "#313244")
      ("BASE03" . "#45475A") ("BASE04" . "#585B70") ("BASE05" . "#CDD6F4")
      ("BASE06" . "#F5E0DC") ("BASE07" . "#B4BEFE") ("BASE08" . "#F38BA8")
      ("BASE09" . "#FAB387") ("BASE0A" . "#F9E2AF") ("BASE0B" . "#A6E3A1")
      ("BASE0C" . "#94E2D5") ("BASE0D" . "#89B4FA") ("BASE0E" . "#CBA6F7")
      ("BASE0F" . "#F2CDCD") ("PINK" . "#F5C2E7")))
#+end_src

#+begin_src emacs-lisp
  (defmacro my/with-palette (palette &rest body)
    (declare (indent 1))
    `(cl-destructuring-bind (&key bg bg-alt dim muted comment fg rosewater lavender
                                  red orange yellow green teal blue mauve flamingo pink
                                  &allow-other-keys)
         ,palette
       ,@body))
#+end_src

Reading the file never signals: any problem yields =nil= and the fallback is
used.

#+begin_src emacs-lisp
  (defun my/theme-state ()
    (when (file-readable-p my/theme-file)
      (condition-case nil
          (with-temp-buffer
            (insert-file-contents my/theme-file)
            (let ((data (json-parse-buffer :object-type 'hash-table)))
              (and (hash-table-p data) data)))
        (error nil))))
#+end_src

The AMOLED setting normally comes from the theme file. =my/amoled-override=
lets a command force it on or off without touching that file. Light themes
always ignore it, so the background can never turn black under dark text.

#+begin_src emacs-lisp
  (defvar my/amoled-override nil
    "Overrides the AMOLED setting of the theme file: `on', `off' or nil.
Nil follows the file.")
#+end_src

#+begin_src emacs-lisp
  (defun my/palette (state)
    (let* ((table (and state (gethash "colors" state)))
           (custom (when (hash-table-p table)
                     (cl-loop for name being the hash-keys of table using (hash-values value)
                              when (and (stringp value)
                                        (string-match-p "\\`#[[:xdigit:]]\\{6\\}\\'" value))
                              collect (cons name value))))
           (colors (append custom my/theme-fallback))
           (light (and state (equal (gethash "variant" state) "light")))
           (file-amoled (and state (eq (gethash "amoled" state) t)))
           (amoled (and (not light)
                        (pcase my/amoled-override
                          ('on t)
                          ('off nil)
                          (_ file-amoled)))))
      (cl-flet ((color (name) (cdr (assoc name colors))))
        (list :bg (if amoled "#000000" (color "BASE00"))
              :bg-alt (if amoled "#0A0A0A" (color "BASE01"))
              :dim (color "BASE02")
              :muted (color "BASE03")
              :comment (color "BASE04")
              :fg (color "BASE05")
              :rosewater (color "BASE06")
              :lavender (color "BASE07")
              :red (color "BASE08")
              :orange (color "BASE09")
              :yellow (color "BASE0A")
              :green (color "BASE0B")
              :teal (color "BASE0C")
              :blue (color "BASE0D")
              :mauve (color "BASE0E")
              :flamingo (color "BASE0F")
              :pink (color "PINK")))))
#+end_src

**** Org

Org buffers are proportional, so only faces that must stay aligned (code,
tables, dates) inherit =fixed-pitch=. Source block delimiter lines use the
block background as their text color, which turns them into blank padding.

*** Reload and watch

A reload rebuilds the theme from the file, then refreshes everything that
caches colors (cursors, SVG tags, the mode line) and applies the opacity. A
file watch triggers it shortly after the last change, which also covers tools
that replace the file atomically.

#+begin_src emacs-lisp

    '(my/theme-base-faces my/theme-ui-faces my/theme-syntax-faces
      my/theme-package-faces my/theme-modeline-faces my/theme-punch-faces
      my/theme-evil-faces my/theme-org-heading-faces my/theme-org-text-faces
      my/theme-org-task-faces my/theme-org-agenda-faces))

  (defun my/theme-faces (palette)
    (apply #'append
           (mapcar (lambda (builder) (funcall builder palette))
                   my/theme-face-builders)))
#+end_src

#+begin_src emacs-lisp

      (dolist (spec `((normal ,fg box)
                      (visual ,comment box)
                      (operator ,blue (bar . 6))
                      (insert ,lavender (bar . 2))
                      (replace ,orange hbar)

  (defun my/theme-reload ()
    (interactive)

           (palette (my/palette state))

** Mode line

=punch-line= draws the mode line. Nix builds it from GitHub, so this section
only configures it. It is loaded when the first graphical frame appears,
because it computes faces from the current frame, and terminal frames keep the
stock mode line. Segments that need the network or other platforms are
switched off.

*** Diagnostics segment

=punch-line= only knows Flycheck, while this configuration uses Flymake. These
functions replace its diagnostics segment with Flymake counts and cache them
briefly per buffer.

#+begin_src emacs-lisp
  (defun my/punch-diagnostics-part (count icon face)
    (when (> count 0)
      (propertize (format "%s %d" (nerd-icons-codicon icon) count)
                  'face `(:inherit ,face))))

  (defun my/punch-diagnostics-string ()
    (let ((errors 0) (warnings 0) (notes 0))
      (dolist (diagnostic (flymake-diagnostics))
        (pcase (flymake-diagnostic-type diagnostic)
          (:error (cl-incf errors))
          (:warning (cl-incf warnings))
          (_ (cl-incf notes))))
      (let ((parts (delq nil (list (my/punch-diagnostics-part notes "nf-cod-lightbulb" 'success)
                                   (my/punch-diagnostics-part warnings "nf-cod-warning" 'warning)
                                   (my/punch-diagnostics-part errors "nf-cod-error" 'error)))))
        (when parts
          (string-join parts " ")))))
#+end_src

Next and previous buffer skip starred buffers. The loop stops once it has

        (funcall change))))

** SVG tags

Replaces plain Org text (stars, keywords, priorities, progress cookies,
timestamps, tags, header lines and property drawers) with SVG images through
=svg-tag-mode=. It needs a graphical frame with SVG support; otherwise the
plain text stays. If a tag fails to draw, the plain text is shown and the error
goes to =*Messages*=. Moving the cursor onto a tag shows its text again for
editing.

*** Metrics

Sizes are derived from the default font. The tunables at the top adjust text
size, padding, line width, vertical centering and the width of progress bars.

#+begin_src emacs-lisp
  (defvar my/svg-text-scale 0.85)
  (defvar my/svg-title-scale 1.5)
  (defvar my/svg-padding 0.7)
  (defvar my/svg-stroke 1.5)
  (defvar my/svg-cap-height 0.72)
  (defvar my/svg-progress-chars 12.0
    "Minimum width, in characters, of a pill that shows a progress bar.")
  (defvar my/svg-metrics-cache (make-hash-table :test #'equal))
#+end_src

#+begin_src emacs-lisp
  (defun my/svg-metrics (&optional scale)
    (let* ((scale (or scale my/svg-text-scale))
           (family (face-attribute 'default :family nil 'default))
           (height (face-attribute 'default :height nil 'default))
           (points (max 6 (round (* (/ height 10.0) scale))))
           (key (cons family points)))
      (unless (and (stringp family) (numberp height))
        (error "svg tag: no usable default font"))
      (or (gethash key my/svg-metrics-cache)
          (let ((info (font-info (format "%s-%d" family points))))
            (unless info
              (error "svg tag: font not found: %s" family))
            (puthash key
                     (list :family family
                           :size (aref info 2)
                           :char (let ((width (aref info 11)))
                                   (if (> width 0) width (round (* 0.55 (aref info 2))))))
                     my/svg-metrics-cache)))))
#+end_src

#+begin_src emacs-lisp
  (defun my/svg-color (face)
    (svg-lib-convert-color (face-attribute face :foreground nil 'default)))

  (defun my/svg-height ()
    (float (min (frame-char-height)
                (round (* (plist-get (my/svg-metrics) :size) 1.5)))))

  (defun my/svg-pill-width (label &optional progress)
    "Return the width of a pill for LABEL.
A pill with PROGRESS is at least `my/svg-progress-chars' characters wide."
    (let* ((char (plist-get (my/svg-metrics) :char))
           (text (+ (* (length (string-trim label)) char) (* 2 my/svg-padding char))))
      (if progress
          (max text (* my/svg-progress-chars char))
        text)))
#+end_src

*** Shapes

A pill is a rounded outline or filled shape with centered text and an optional
progress bar. Either end can be round or square so two pills can be joined.

#+begin_src emacs-lisp
  (defun my/svg-path (x0 y0 x1 y1 rl rr &optional open-left)
    (if open-left
        (format "M%f,%f H%f A%f,%f 0 0 1 %f,%f V%f A%f,%f 0 0 1 %f,%f H%f"
                x0 y0 (- x1 rr)
                rr rr x1 (+ y0 rr) (- y1 rr)
                rr rr (- x1 rr) y1 x0)
      (format "M%f,%f H%f A%f,%f 0 0 1 %f,%f V%f A%f,%f 0 0 1 %f,%f H%f A%f,%f 0 0 1 %f,%f V%f A%f,%f 0 0 1 %f,%f Z"
              (+ x0 rl) y0 (- x1 rr)
              rr rr x1 (+ y0 rr) (- y1 rr)
              rr rr (- x1 rr) y1 (+ x0 rl)
              rl rl x0 (- y1 rl) (+ y0 rl)
              rl rl (+ x0 rl) y0)))
#+end_src

#+begin_src emacs-lisp
  (cl-defun my/svg-draw-pill (svg label face x &key filled (left 'round) (right 'round)
                                  progress open-left)
    "Draw LABEL on SVG as a pill whose left edge is at X and return its width."
    (let* ((label (string-trim label))
           (metrics (my/svg-metrics))
           (size (plist-get metrics :size))
           (height (my/svg-height))
           (width (my/svg-pill-width label progress))
           (radius (* height 0.28))
           (rl (if (eq left 'round) radius 0))
           (rr (if (eq right 'round) radius 0))
           (stroke my/svg-stroke)
           (color (my/svg-color face))
           (background (svg-lib-convert-color
                        (face-attribute 'default :background nil 'default))))
      (if filled
          (svg-node svg 'path :d (my/svg-path x 0 (+ x width) height rl rr) :fill color)
        (let ((inset (/ stroke 2.0)))
          (svg-node svg 'path
                    :d (my/svg-path (if open-left x (+ x inset)) inset
                                    (- (+ x width) inset) (- height inset)
                                    (max 0 (- rl inset)) (max 0 (- rr inset))
                                    open-left)
                    :fill "none" :stroke color :stroke-width stroke)))
      (when (and progress (> progress 0))
        (let ((inner (max 0 (- radius stroke))))
          (svg-node svg 'path
                    :d (my/svg-path (+ x stroke) stroke
                                    (+ x stroke (* progress (- width (* 2 stroke))))
                                    (- height stroke)
                                    inner (if (>= progress 1) inner 0))
                    :fill color :fill-opacity 0.35)))
      (svg-text svg label
                :font-family (plist-get metrics :family)
                :font-size size
                :font-weight (if filled 700 600)
                :fill (if filled background color)
                :text-anchor "middle"
                :x (+ x (/ width 2.0))
                :y (+ (/ height 2.0) (* size my/svg-cap-height 0.5)))
      width))
#+end_src

*** Images

Builders that turn a piece of text into an image: single pills, joined pairs,
heading markers and header lines.

#+begin_src emacs-lisp
  (cl-defun my/svg-pill (label face &key filled (left 'round) (right 'round)
                               progress open-left (gap 0))
    (let* ((char (plist-get (my/svg-metrics) :char))
           (svg (svg-create (+ (my/svg-pill-width label progress) (* gap char))
                            (my/svg-height))))
      (my/svg-draw-pill svg label face 0 :filled filled :left left :right right
                        :progress progress :open-left open-left)
      (svg-image svg :ascent 'center)))

  (defun my/svg-pair (first second face filled-second)
    "Return one image of FIRST (outlined) joined to SECOND (filled or outlined)."
    (let* ((first-width (my/svg-pill-width first))
           (svg (svg-create (+ first-width (my/svg-pill-width second))
                            (my/svg-height))))
      (my/svg-draw-pill svg first face 0 :right 'square)
      (my/svg-draw-pill svg second face first-width
                        :filled filled-second :left 'square
                        :open-left (not filled-second))
      (svg-image svg :ascent 'center)))
#+end_src

Heading markers depend on the level: dot, ring, diamond, outlined diamond,
then a small dot.

#+begin_src emacs-lisp
  (defun my/svg-diamond (cx cy radius)
    (list (cons cx (- cy radius)) (cons (+ cx radius) cy)
          (cons cx (+ cy radius)) (cons (- cx radius) cy)))

  (defun my/svg-star (level)
    "Return the marker for a heading of LEVEL."
    (let* ((height (my/svg-height))
           (width (float (frame-char-width)))
           (cx (/ width 2.0))
           (cy (/ height 2.0))
           (radius (* height 0.24))
           (stroke my/svg-stroke)
           (inset (/ stroke 2.0))
           (color (my/svg-color (intern (format "org-level-%d" (1+ (mod (1- level) 8))))))
           (svg (svg-create width height)))
      (pcase level
        (1 (svg-circle svg cx cy radius :fill color))
        (2 (svg-circle svg cx cy (- radius inset)
                       :fill "none" :stroke color :stroke-width stroke))
        (3 (svg-polygon svg (my/svg-diamond cx cy (* radius 1.15)) :fill color))
        (4 (svg-polygon svg (my/svg-diamond cx cy (- (* radius 1.15) inset))
                        :fill "none" :stroke color :stroke-width stroke))
        (_ (svg-circle svg cx cy (* radius 0.55) :fill color)))
      (svg-image svg :ascent 'center)))
#+end_src

Header lines: the title is a large bold text after an accent bar. The date and

                     :rx (/ bar 2) :fill color)
      (svg-text svg text
                :font-family (plist-get metrics :family)
                :font-size size
                :font-weight 700
                :fill color
                :x indent
                :y (+ (/ height 2.0) (* size my/svg-cap-height 0.5)))
      (svg-image svg :ascent 'center)))
#+end_src

*** Entry helpers

An entry is a regexp whose first group is the text to replace, plus a function
that returns the image for it. Entries never signal: on any error the plain
text is kept.

#+begin_src emacs-lisp
  (defconst my/svg-date-re "[0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}")
  (defconst my/svg-tag-char "[[:alnum:]_@#%]")

  (defun my/svg-entry (regexp function)
    "Return an `svg-tag-tags' entry. If FUNCTION fails, the plain text is shown."
    (cons regexp
          (list (lambda (tag)
                  (condition-case err
                      (save-match-data (funcall function tag))
                    (error (message "svg tag: %s" (error-message-string err))
                           tag))))))
#+end_src

#+begin_src emacs-lisp
  (defun my/svg-ci (word)

               word ""))

*** Property drawers

A property drawer is drawn as a small "properties" label, one joined key and
value label per property, and a hidden closing line. Only real property
drawers are touched: a line counts only when nothing but property lines
separate it from the opening line, so other drawers and text inside blocks keep
their plain text. This covers Org-roam =ID= drawers as well. Drawers are
hidden by default (see Property drawers), so these labels show while they are
revealed.

#+begin_src emacs-lisp
  (defun my/svg-in-property-drawer-p (pos &optional from-above)
    "Return the opening line's position when the line at POS is in a property drawer.
Only property lines may sit between it and the opening line above; otherwise
return nil. With FROM-ABOVE, start at the previous line, which is how the
closing line is checked."
    (save-excursion
      (goto-char pos)
      (forward-line 0)
      (when from-above
        (forward-line -1))
      (let ((case-fold-search t))
        (while (and (not (bobp))
                    (not (looking-at-p "[ \t]*:\\(?:PROPERTIES\\|END\\):"))
                    (looking-at-p "[ \t]*:\\S-+?:\\(?:[ \t]\\|$\\)"))
          (forward-line -1))
        (and (looking-at-p "[ \t]*:PROPERTIES:[ \t]*$") (point)))))
#+end_src

** Property drawers

Property drawers are hidden completely, including their folding ellipsis, so
the title gets the top of the buffer. Each drawer is covered by an invisible
overlay that is rebuilt when the buffer opens and after every save. A drawer is
only covered when nothing but property lines sit between its first and last
line, so a broken drawer can never swallow the text below it. The toggle in
Functions reveals them again.

#+begin_src emacs-lisp
  (defun my/org-hide-property-drawers ()
    "Cover every property drawer in the buffer with an invisible overlay."
    (when (derived-mode-p 'org-mode)
      (remove-overlays (point-min) (point-max) 'my-drawer t)
      (save-excursion
        (save-match-data
          (goto-char (point-min))
          (while (re-search-forward "^[ \t]*:PROPERTIES:[ \t]*$" nil t)
            (let ((start (line-beginning-position)))
              (when (and (not (my/svg-in-block-p start))
                         (re-search-forward "^[ \t]*:END:[ \t]*$" nil t)
                         (eql (my/svg-in-property-drawer-p (line-beginning-position) t)
                              start))
                (let ((overlay (make-overlay start (min (1+ (point)) (point-max))
                                             nil t)))
                  (overlay-put overlay 'my-drawer t)
                  (overlay-put overlay 'invisible 'my-drawer)
                  (overlay-put overlay 'evaporate t)))))))))
#+end_src
